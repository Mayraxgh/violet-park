from machine import Pin, TouchPad, time_pulse_us, reset from time import sleep, sleep_ms, sleep_us, ticks_ms, ticks_diff import network import ujson import urequests import ntptime import time import gc # ========================================== # VIOLET PARK - ESP32 + FIREBASE # ========================================== # ---- CAMBIA ESTOS DATOS (no los compartas) ---- WIFI_SSID = "TU_WIFI" WIFI_PASSWORD = "TU_PASSWORD_WIFI" API_KEY = "TU_API_KEY" DATABASE_URL = "https://violet-park-default-rtdb.firebaseio.com" FIREBASE_EMAIL = "TU_CORREO" FIREBASE_PASSWORD = "TU_PASSWORD_FIREBASE" # ========================================== # PINES # ========================================== trig = Pin(5, Pin.OUT) echo = Pin(18, Pin.IN) led_rojo = Pin(13, Pin.OUT) led_blanco = Pin(32, Pin.OUT) led_amarillo = Pin(25, Pin.OUT) led_verde = Pin(26, Pin.OUT) buzzer = Pin(19, Pin.OUT) touch = TouchPad(Pin(33)) # ========================================== # CONFIGURACION # ========================================== # None = se calibra solo al arrancar (no toques el cable en ese momento). # Si no detecta tu toque, pon un numero fijo, por ejemplo 400. UMBRAL_TOUCH = None HISTERESIS_TOUCH = 25 DEBUG_TOUCH = False # True para ver los valores del sensor tactil DIST_SEGURA = 20 DIST_PRECAUCION = 10 DIST_PELIGRO = 3 INTERVALO_PARPADEO = 300 # ms PERIODO_LECTURA = 150 # ms PERIODO_ACTUAL = 1000 # ms (dato en tiempo real) PERIODO_HISTORIAL = 5000 # ms (lotes de cambios) RENOVAR_TOKEN = 50 * 60 * 1000 # MicroPython en ESP32 cuenta desde el anio 2000; Firebase/JS usan 1970 EPOCH_OFFSET = 946684800 if time.gmtime(0)[0] == 2000 else 0 # ========================================== # VARIABLES # ========================================== wlan = network.WLAN(network.STA_IF) umbral_touch = 400 touch_activo = False touch_valor = 0 touch_anterior = False estado_alerta = 0 ultimo_parpadeo = ticks_ms() estado_anterior = None datos_actuales = {} pendientes = [] token = None token_ms = 0 hora_ok = False pulso = 0 inicio_ms = ticks_ms() ultimo_chequeo = ticks_ms() # ========================================== # APAGAR COMPONENTES # ========================================== def apagar_todo(): trig.value(0) led_rojo.value(0) led_blanco.value(0) led_amarillo.value(0) led_verde.value(0) buzzer.value(0) apagar_todo() # ========================================== # WIFI, HORA Y FIREBASE # ========================================== def conectar_wifi(espera=20000): wlan.active(True) if wlan.isconnected(): return True print("Conectando a Wi-Fi...") wlan.connect(WIFI_SSID, WIFI_PASSWORD) inicio = ticks_ms() while not wlan.isconnected(): if ticks_diff(ticks_ms(), inicio) > espera: print("Wi-Fi no disponible") return False sleep_ms(300) print("Wi-Fi conectado:", wlan.ifconfig()[0]) return True def sincronizar_hora(): global hora_ok try: ntptime.host = "pool.ntp.org" ntptime.settime() hora_ok = True print("Hora sincronizada") except Exception as e: print("No se pudo sincronizar la hora:", e) def ahora_unix(): return (time.time() + EPOCH_OFFSET) if hora_ok else 0 def autenticar_firebase(): global token, token_ms url = ( "https://identitytoolkit.googleapis.com/v1/" "accounts:signInWithPassword?key=" + API_KEY ) datos = { "email": FIREBASE_EMAIL, "password": FIREBASE_PASSWORD, "returnSecureToken": True } r = None try: gc.collect() r = urequests.post( url, json=datos, headers={"Content-Type": "application/json"} ) resultado = r.json() if r.status_code != 200: print("Error autenticando:", resultado) return False token = resultado["idToken"] token_ms = ticks_ms() print("Firebase conectado correctamente") return True except Exception as e: print("Error de autenticacion Firebase:", e) return False finally: if r is not None: r.close() gc.collect() def firebase_enviar(metodo, ruta, datos): global token if token is None: return False r = None try: gc.collect() url = DATABASE_URL + "/" + ruta + ".json?auth=" + token cuerpo = ujson.dumps(datos) cabecera = {"Content-Type": "application/json"} if metodo == "PUT": r = urequests.put(url, data=cuerpo, headers=cabecera) else: r = urequests.post(url, data=cuerpo, headers=cabecera) if r.status_code == 401: token = None print("Token vencido, se renovara") return False if r.status_code not in (200, 201, 204): print("Error HTTP:", r.status_code) return False return True except Exception as e: print("Error Firebase:", e) return False finally: if r is not None: r.close() gc.collect() # ========================================== # SENSOR TACTIL # ========================================== def calibrar_touch(): global umbral_touch if UMBRAL_TOUCH is not None: umbral_touch = UMBRAL_TOUCH return total = 0 for _ in range(20): total += touch.read() sleep_ms(20) base = total // 20 umbral_touch = base - max(40, base * 15 // 100) print("Touch base:", base, "| umbral:", umbral_touch) def leer_touch(): global touch_activo, touch_valor try: v = (touch.read() + touch.read() + touch.read()) // 3 except Exception: return touch_valor = v # Histeresis: evita que el valor "parpadee" cerca del umbral if v < umbral_touch: touch_activo = True elif v > umbral_touch + HISTERESIS_TOUCH: touch_activo = False if DEBUG_TOUCH: print("touch:", v, "activo:", touch_activo) # ========================================== # SENSOR ULTRASONICO # ========================================== def medir_distancia(): lecturas = [] for _ in range(3): trig.value(0) sleep_us(2) trig.value(1) sleep_us(10) trig.value(0) try: duracion = time_pulse_us(echo, 1, 30000) if duracion > 0: d = duracion / 58.0 if 2 <= d <= 400: lecturas.append(d) except OSError: pass sleep_ms(25) if len(lecturas) < 2: return None lecturas.sort() return round(lecturas[len(lecturas) // 2], 1) # ========================================== # ACTUALIZAR COMPONENTES Y REGISTRAR CAMBIOS # ========================================== def actualizar_sensores(): global estado_alerta, ultimo_parpadeo global datos_actuales, estado_anterior, touch_anterior ahora = ticks_ms() leer_touch() led_blanco.value(1 if touch_activo else 0) distancia = medir_distancia() led_verde.value(0) led_amarillo.value(0) led_rojo.value(0) buzzer.value(0) if distancia is None: estado = "SIN_LECTURA" estado_alerta = 0 elif distancia > DIST_SEGURA: estado = "SEGURO" led_verde.value(1) estado_alerta = 0 elif distancia > DIST_PRECAUCION: estado = "PRECAUCION" led_amarillo.value(1) estado_alerta = 0 elif distancia > DIST_PELIGRO: estado = "ALERTA" if ticks_diff(ahora, ultimo_parpadeo) >= INTERVALO_PARPADEO: ultimo_parpadeo = ahora estado_alerta = 1 - estado_alerta led_rojo.value(estado_alerta) buzzer.value(estado_alerta) else: estado = "PELIGRO" led_rojo.value(1) buzzer.value(1) estado_alerta = 1 datos_actuales = { "distancia": distancia, "estado": estado, "led_rojo": bool(led_rojo.value()), "led_blanco": bool(led_blanco.value()), "led_amarillo": bool(led_amarillo.value()), "led_verde": bool(led_verde.value()), "buzzer": bool(buzzer.value()), "tactil": touch_activo, "touch_valor": touch_valor } # Registrar cambios de estado Y contactos tactiles if distancia is not None: evento = None if estado != estado_anterior: evento = "ESTADO" elif touch_activo != touch_anterior: evento = "CONTACTO" if touch_activo else "LIBERADO" if evento: pendientes.append({ "segundos": ahora_unix(), "distancia": distancia, "estado": estado, "evento": evento, "led_rojo": bool(led_rojo.value()), "led_blanco": bool(led_blanco.value()), "led_amarillo": bool(led_amarillo.value()), "led_verde": bool(led_verde.value()), "buzzer": bool(buzzer.value()), "tactil": touch_activo }) if len(pendientes) > 50: pendientes.pop(0) print(evento, estado, distancia, "cm | tactil:", touch_activo) estado_anterior = estado touch_anterior = touch_activo # ========================================== # ENVIAR A FIREBASE # ========================================== def enviar_actual(): global pulso if not datos_actuales: return pulso += 1 datos_actuales["pulso"] = pulso datos_actuales["actualizacion"] = ahora_unix() datos_actuales["uptime"] = ticks_diff(ticks_ms(), inicio_ms) // 1000 try: datos_actuales["wifi_rssi"] = wlan.status("rssi") except Exception: pass firebase_enviar("PUT", "violetPark/actual", datos_actuales) def enviar_historial(): if not pendientes: return lote = pendientes[:] if firebase_enviar("POST", "violetPark/historial", lote): del pendientes[:len(lote)] print("Historial guardado:", len(lote), "registros") else: print("Registros conservados para reintentar") def mantener_conexion(ahora): global ultimo_chequeo if ticks_diff(ahora, ultimo_chequeo) < 15000: return ultimo_chequeo = ahora if not wlan.isconnected(): print("Wi-Fi perdido, reconectando...") conectar_wifi(8000) return if not hora_ok: sincronizar_hora() if token is None or ticks_diff(ahora, token_ms) > RENOVAR_TOKEN: autenticar_firebase() # ========================================== # PROGRAMA PRINCIPAL # ========================================== def main(): calibrar_touch() if conectar_wifi(): sincronizar_hora() autenticar_firebase() print("Violet Park en marcha") t_lectura = ticks_ms() t_actual = ticks_ms() t_historial = ticks_ms() touch_enviado = False while True: ahora = ticks_ms() if ticks_diff(ahora, t_lectura) >= PERIODO_LECTURA: actualizar_sensores() t_lectura = ahora if token is not None and wlan.isconnected(): # El contacto tactil se envia de inmediato if (touch_activo != touch_enviado or ticks_diff(ahora, t_actual) >= PERIODO_ACTUAL): enviar_actual() touch_enviado = touch_activo t_actual = ticks_ms() if (pendientes and ticks_diff(ahora, t_historial) >= PERIODO_HISTORIAL): enviar_historial() t_historial = ticks_ms() mantener_conexion(ahora) sleep_ms(10) try: main() except Exception as error: print("Error en Violet Park:", error) apagar_todo() sleep(5) reset() finally: apagar_todo()
